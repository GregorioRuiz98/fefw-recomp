set(FEFW_MODULE_TARGETS "")
foreach(module main sdk subsdk0 subsdk1 subsdk2 rtld)
    file(GLOB sources CONFIGURE_DEPENDS "${FEFW_LIFT_DIR}/${module}/c_*.cpp")
    set(descriptor "${FEFW_LIFT_DIR}/${module}/module_${module}.cpp")
    if(NOT sources OR NOT EXISTS "${descriptor}")
        message(FATAL_ERROR "Missing generated ${module}: run tools/translate.py lift first")
    endif()
    add_library(fefw_${module} STATIC ${sources} "${descriptor}")
    target_include_directories(fefw_${module} PUBLIC "${FEFW_ENGINE}/recomp/include" "${FEFW_LIFT_DIR}/${module}")
    target_compile_options(fefw_${module} PRIVATE ${FEFW_LIFT_OPT} /w /EHs-c- /GR- -fno-strict-aliasing -ffp-contract=off -mcx16)
    list(APPEND FEFW_MODULE_TARGETS fefw_${module})
endforeach()
